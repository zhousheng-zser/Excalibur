cmake_minimum_required(VERSION 3.0.0)
project(Excalibur C CXX)

option(x86 "This option is necessary for x86 platform" ON)
option(USE_CUDA "Build Excalibur with CUDA support" OFF)
option(USE_CUDNN "Build Excalibur with cuDNN library support" OFF)
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(INT8_DATA "Use int8 quantilized model" OFF)
option(HALF_DATA "Use int16 quantilized model" OFF)
option(USE_OPENMP "Use OpenMP" OFF)
option(COSINE_DISTANCE "Use cosine distance" OFF)

include(Dependencies.cmake)
set(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
if(UNIX OR APPLE)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIC -Wall")
endif()

add_subdirectory(Julius)
add_subdirectory(Excalibur)
add_subdirectory(Damocles)
add_subdirectory(Romancia)
add_subdirectory(Longinus)
add_subdirectory(Irisvian)
add_subdirectory(Cassius)
