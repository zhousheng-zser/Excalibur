cmake_minimum_required(VERSION 3.14.3)
project(Excalibur C CXX)

option(WITH_JNI "Build jni interface" OFF)
option(RELEASE_SDK "Build external sdk" ON)
option(TRIAL "Build trail sdk" OFF)
option(USE_CUDA "Build Excalibur with CUDA support" OFF)
option(USE_CUDNN "Build Excalibur with cuDNN library support" OFF)
option(USE_OPENCV "Build with opencv library support" OFF)
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(USE_OPENMP "Use OpenMP" ON)
option(COSINE_DISTANCE "Use cosine distance" ON)
option(FMT_HEADER_ONLY "Use header-only libfmt" ON)
option(HAVE_SSE "CPU SSE support" ON)
option(HAVE_AVX "CPU AVX support" ON)
option(HAVE_AVX2 "CPU AVX2 support" ON)

set(CMAKE_CXX_STANDARD 17)

set(PRIMITIVES_TARGET_NAME Primitives)
get_filename_component(GLASSSIX_EXCALIBUR_JNI ../jni/jni ABSOLUTE)
get_filename_component(GLASSSIX_EXCALIBUR_INCLUDE ../include ABSOLUTE)
get_filename_component(GLASSSIX_EXCALIBUR_SOURCE ../src ABSOLUTE)

if(WITH_JNI)
set(JNI_INCLUDE /home/build_slave/jdk1.8.0_221/include 
		/home/build_slave/jdk1.8.0_221/include/linux)
endif()

if(HAVE_AVX2)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mavx2")
	add_definitions(-DHAVE_AVX2)
endif()
if(HAVE_AVX)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mavx -mfma")
	add_definitions(-DHAVE_AVX)
endif()
if(HAVE_SSE)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mmmx -msse -msse2 -msse3 -mssse3 -msse4.1 -msse4.2 -mf16c")
	add_definitions(-DHAVE_SSE)
endif()

if(RELEASE_SDK)
	add_definitions(-DRELEASE_SDK)
endif()

if(TRIAL)
	add_definitions(-DTRIAL)
endif()

if(USE_OPENCV)
	add_definitions(-DUSE_OPENCV)
endif()

if(CMAKE_BUILD_TYPE STREQUAL "")
	set(CMAKE_BUILD_TYPE "Release")
endif()
message(STATUS "CMAKE_BUILD_TYPE: ${CMAKE_BUILD_TYPE}")

set(QUANTIZATION_TYPE "SINGLE" CACHE STRING "Selected quantization type for Cassius library")
set_property(CACHE QUANTIZATION_TYPE PROPERTY STRINGS "SINGLE;HALF;INT8")

if(QUANTIZATION_TYPE STREQUAL "HALF" OR QUANTIZATION_TYPE STREQUAL "half")
	add_definitions(-DHALF_DATA)
elseif(QUANTIZATION_TYPE STREQUAL "INT8" OR QUANTIZATION_TYPE STREQUAL "int8")
        add_definitions(-DINT8_DATA)
else()
	set(QUANTIZATION_TYPE "SINGLE")
endif()
message(STATUS "QUANTIZATION_TYPE: ${QUANTIZATION_TYPE}")


include(Dependencies.cmake)
set(LIBRARY_OUTPUT_PATH ${PROJECT_BINARY_DIR}/lib)
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/bin)
if(UNIX OR APPLE)
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIC -Wall -std=c++17 -fpermissive")
endif()
