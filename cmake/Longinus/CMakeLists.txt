	file(GLOB Longinus_jni_source_files ${CMAKE_CURRENT_SOURCE_DIR}/../../jni/jni/Longimila/*.cpp)

	if(WITH_JNI)

	else()
		add_library(Longinus ${Longinus_source_files})
	endif()
endif()
